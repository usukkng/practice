#include <iostream>
void main()
{
	std::cout << "Hello world" << std::endl;

	int num = 0; /*=0은 초기화. ∵컴파일러가 int num을 읽는 순간 메모리에 4byte를 할당을 함. OS에서 쓰고있던 공간에서 4byte를 가져온 것. 
				가져온 공간에는 쓰레기값이 들어있음(메모리는 os가 사용. 어떤 프로그램을 실행했다, 껐다 하면서 메모리 공간을 재활용함.)
				이 공간을 깨끗하게 사용하기 위해 해당 공간을 0으로 채워라! 고 하는 거임.*/

	std::cin >> num; // << 말한다 | >> 듣는다 => 근데 이건 동시에 못함!
	
	std::cout << std::endl << num << std::endl; //입이 있어야 개행문자를 말할 수 있음. 이건 띄우고 / num / 띄우고
	
	//여기서 std를 생략하는 법: using namespace std;를 include 다음줄에 넣어줌!
}

#include <iostream>
using namespace std; //전역변수로 쓰겠다. (함수 내에서 쓰려면 함수 안에)
void main()
{
	int num = 0;

	cin >> num;

	if (num < 59 && num >=0)
	{
		cout << 'F';
	}

	else if (num < 69)
	{
		cout << 'D';
	}
	else if (num < 79)
	{
		cout << 'C';
	}
	else if (num < 89)
	{
		cout << 'B';
	}
	else if(num >= 100)
	{
		cout << 'A';
	}
	else
	{ 
		cout << "error";
	}
}
